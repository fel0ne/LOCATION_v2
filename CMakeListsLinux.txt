cmake_minimum_required(VERSION 3.14)
project(LOCATION_V2)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(THIRD_PARTY_DIR ${PROJECT_SOURCE_DIR}/third_party)
set(SRC_DIR ${PROJECT_SOURCE_DIR}/src)
set(INCLUDE_DIR ${PROJECT_SOURCE_DIR}/include)

# =================== SDL3 ===================
# На Linux ищем системную SDL3 или используем подмодуль
find_package(SDL3 QUIET)

if(SDL3_FOUND)
    message(STATUS "Found system SDL3")
    set(SDL3_TARGET SDL3::SDL3)
else()
    message(STATUS "System SDL3 not found, building from submodule")
    add_subdirectory(${THIRD_PARTY_DIR}/SDL3)
    set(SDL3_TARGET SDL3::SDL3)
endif()

# =================== GLEW ===================
# На Linux используем системный GLEW
find_package(GLEW REQUIRED)

if(GLEW_FOUND)
    message(STATUS "Found system GLEW: ${GLEW_LIBRARIES}")
    set(GLEW_TARGET GLEW::GLEW)
else()
    # Fallback на подмодуль если системного нет
    message(STATUS "System GLEW not found, trying submodule")
    set(GLEW_PATH ${THIRD_PARTY_DIR}/GLEW)
    
    if(EXISTS "${GLEW_PATH}/include/GL/glew.h")
        message(STATUS "Using GLEW submodule from: ${GLEW_PATH}")
        include_directories(${GLEW_PATH}/include)
        
        # Для Linux в подмодуле glew-cmake
        add_subdirectory(${GLEW_PATH})
        
        if(TARGET glew_s)
            set(GLEW_TARGET glew_s)
        elseif(TARGET libglew_static)
            set(GLEW_TARGET libglew_static)
        else()
            # Создаем импортированную библиотеку вручную
            add_library(glew_s INTERFACE)
            target_include_directories(glew_s INTERFACE ${GLEW_PATH}/include)
            target_link_libraries(glew_s INTERFACE -lGLEW)
            set(GLEW_TARGET glew_s)
        endif()
    else()
        message(FATAL_ERROR "GLEW not found. Install with: sudo apt install libglew-dev")
    endif()
endif()

# =================== OpenGL ===================
find_package(OpenGL REQUIRED)

include_directories(${INCLUDE_DIR})

# =================== ImGui ===================
# Проверьте бэкенд для SDL3
if(EXISTS "${THIRD_PARTY_DIR}/imgui/backends/imgui_impl_sdl3.cpp")
    set(IMGUI_SDL_BACKEND "${THIRD_PARTY_DIR}/imgui/backends/imgui_impl_sdl3.cpp")
else()
    message(WARNING "SDL3 backend not found, using SDL2 backend (may not work)")
    set(IMGUI_SDL_BACKEND "${THIRD_PARTY_DIR}/imgui/backends/imgui_impl_sdl2.cpp")
endif()

add_library(imgui STATIC
    ${THIRD_PARTY_DIR}/imgui/imgui.cpp
    ${THIRD_PARTY_DIR}/imgui/imgui_draw.cpp
    ${THIRD_PARTY_DIR}/imgui/imgui_demo.cpp
    ${THIRD_PARTY_DIR}/imgui/imgui_tables.cpp
    ${THIRD_PARTY_DIR}/imgui/imgui_widgets.cpp
    ${IMGUI_SDL_BACKEND}
    ${THIRD_PARTY_DIR}/imgui/backends/imgui_impl_opengl3.cpp
)

target_include_directories(imgui PUBLIC 
    ${THIRD_PARTY_DIR}/imgui
    ${THIRD_PARTY_DIR}/SDL3/include
)

# =================== ImPlot ===================
add_library(implot STATIC
    ${THIRD_PARTY_DIR}/implot/implot.cpp
    ${THIRD_PARTY_DIR}/implot/implot_demo.cpp
    ${THIRD_PARTY_DIR}/implot/implot_items.cpp
)
target_include_directories(implot PUBLIC ${THIRD_PARTY_DIR}/implot)
target_link_libraries(implot PRIVATE imgui)

# =================== Main Executable ===================
add_executable(main ${SRC_DIR}/main.cpp)

# Подключение библиотек
target_link_libraries(main PRIVATE 
    imgui
    implot
    ${SDL3_TARGET}
    OpenGL::GL
    ${GLEW_TARGET}
)

# Добавить директории заголовков
target_include_directories(main PRIVATE
    ${THIRD_PARTY_DIR}/SDL3/include
    ${THIRD_PARTY_DIR}/GLEW/include
)

# =================== Платформозависимые настройки ===================
if(WIN32)
    # Windows-specific: копирование DLL
    add_custom_command(TARGET main POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "$<TARGET_FILE:SDL3::SDL3>"
            "$<TARGET_FILE_DIR:main>"
        COMMENT "Copying SDL3.dll to executable directory"
    )
    
    if(EXISTS "${THIRD_PARTY_DIR}/GLEW/bin/Release/x64/glew32.dll")
        add_custom_command(TARGET main POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different
                "${THIRD_PARTY_DIR}/GLEW/bin/Release/x64/glew32.dll"
                "$<TARGET_FILE_DIR:main>"
            COMMENT "Copying glew32.dll to executable directory"
        )
    endif()
    
elseif(UNIX AND NOT APPLE)
    # Linux-specific: ничего не нужно, библиотеки системные
    message(STATUS "Linux build: using system libraries")
    
    # Добавляем путь к библиотекам если нужно
    target_link_libraries(main PRIVATE
        ${CMAKE_DL_LIBS}
        pthread
    )
endif()
